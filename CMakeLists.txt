cmake_minimum_required(VERSION 3.21)
project(image_codec CXX)
add_compile_definitions(_USE_MATH_DEFINES NOMINMAX )
set(CMAKE_CXX_STANDARD 17)
add_executable(llcompc llcompc.cpp llcomp.hpp netpbm.hpp)
add_executable(llcompd llcompd.cpp llcomp.hpp netpbm.hpp)

add_executable(llcompc-xl llcompc.cpp llcomp.hpp netpbm.hpp)
target_compile_definitions(llcompc-xl PRIVATE SELECT_MODEL=ModelSizeLarge)
add_executable(llcompd-xl llcompd.cpp llcomp.hpp netpbm.hpp)
target_compile_definitions(llcompd-xl PRIVATE SELECT_MODEL=ModelSizeLarge)

add_executable(llcompc-s llcompc.cpp llcomp.hpp netpbm.hpp)
target_compile_definitions(llcompc-s PRIVATE SELECT_MODEL=ModelSizeSmall)
add_executable(llcompd-s llcompd.cpp llcomp.hpp netpbm.hpp)
target_compile_definitions(llcompd-s PRIVATE SELECT_MODEL=ModelSizeSmall)

# RC engine (CABAC replaced by the PLT2 back-end on the tiny context): llcompc-rc / llcompd-rc
find_package(Threads REQUIRED)
add_executable(llcompc-rc llcompc_rc.cpp llcomp_rc.hpp llcomp_rc_tables.h netpbm.hpp avx2.h)
add_executable(llcompd-rc llcompd_rc.cpp llcomp_rc.hpp llcomp_rc_tables.h netpbm.hpp avx2.h)
foreach(t llcompc-rc llcompd-rc)
  target_link_libraries(${t} PRIVATE Threads::Threads)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
      target_compile_options(${t} PRIVATE /O3 /QxHost)
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")          # clang-cl understands -march
      target_compile_options(${t} PRIVATE /O2 -march=native)
    else()                                                  # cl.exe ignores -march: without /arch it is SSE2 -> SIMDe path
      set(LLC_MSVC_ARCH "AVX2" CACHE STRING "cl.exe /arch: AVX2 or AVX512")
      target_compile_options(${t} PRIVATE /O2 /Ob3 /arch:${LLC_MSVC_ARCH})
    endif()
  else()
    target_compile_options(${t} PRIVATE -O3 -march=native)
  endif()
endforeach()
